            $labels = array(

                'has_archive'        => true,
                'hierarchical'       => false,
                'menu_position'      => 25,
                'menu_icon'          => 'dashicons-portfolio', // Extensible standard icon font string map

                'supports'           => array( 'title', 'editor', 'thumbnail', 'excerpt', 'custom-fields' ),
